const SUPABASE_CONFIG = {
  url: "COLE_AQUI",   // Project URL, ex.: https://abcdefgh.supabase.co
  key: "COLE_AQUI"    // Publishable key (sb_publishable_...) ou anon key
};
const FIREBASE_CONFIG = {
  apiKey: "COLE_AQUI",
  authDomain: "COLE_AQUI",
  projectId: "COLE_AQUI",
  storageBucket: "COLE_AQUI",
  messagingSenderId: "COLE_AQUI",
  appId: "COLE_AQUI"
};
/* ================================================================================ */
(function(){
  if(window.storage) return; // dentro do Claude, usa o armazenamento nativo
  var usaSupabase = SUPABASE_CONFIG.url !== "COLE_AQUI" && SUPABASE_CONFIG.key !== "COLE_AQUI" && window.supabase;
  var usaFirebase = FIREBASE_CONFIG.apiKey && FIREBASE_CONFIG.apiKey !== "COLE_AQUI" && window.firebase;
  if(usaSupabase){
    var sb = window.supabase.createClient(SUPABASE_CONFIG.url, SUPABASE_CONFIG.key);
    window.storage = {
      get: async function(key){
        var r = await sb.from('kv').select('value').eq('key', key).maybeSingle();
        if(r.error) throw r.error;
        return r.data ? {key: key, value: r.data.value, shared: true} : null;
      },
      set: async function(key, value){
        var r = await sb.from('kv').upsert({key: key, value: value}, {onConflict: 'key'});
        if(r.error) throw r.error;
        return {key: key, value: value, shared: true};
      }
    };
    return;
  }
  if(usaFirebase){
    firebase.initializeApp(FIREBASE_CONFIG);
    var db = firebase.firestore();
    window.storage = {
      get: async function(key){
        var d = await db.collection('ibnv').doc(key).get();
        if(!d.exists) return null;
        return {key: key, value: d.data().value, shared: true};
      },
      set: async function(key, value){
        await db.collection('ibnv').doc(key).set({value: value});
        return {key: key, value: value, shared: true};
      }
    };
    return;
  }
  window.addEventListener('DOMContentLoaded', function(){
    var b = document.createElement('div');
    b.style.cssText = 'background:#FFF3CD;color:#6B4E00;padding:10px 16px;font:14px Arial;text-align:center;';
    b.textContent = 'Banco de dados ainda não configurado: os dados NÃO estão sendo salvos.';
    document.body.insertBefore(b, document.body.firstChild);
  });
})();

function toast(msg){
  try{
    let box = document.getElementById('toastBox');
    if(!box){
      box = document.createElement('div'); box.id = 'toastBox';
      box.style.cssText = 'position:fixed;left:50%;bottom:24px;transform:translateX(-50%);z-index:9999;display:flex;flex-direction:column;gap:8px;align-items:center;max-width:92vw;';
      document.body.appendChild(box);
    }
    const t = document.createElement('div');
    t.textContent = String(msg);
    t.style.cssText = 'background:#231F20;color:#fff;padding:10px 16px;border-radius:8px;font:14px Inter,Arial,sans-serif;box-shadow:0 4px 14px rgba(0,0,0,.25);';
    box.appendChild(t);
    setTimeout(()=>{ t.remove(); }, 4500);
  }catch(e){ console.warn(msg); }
}

function pedirConfirmacao(anchor, texto, acoes){
  if(typeof acoes==='function') acoes = [{label:'Sim, excluir', fn:acoes}];
  const box = document.createElement('div');
  box.className = 'confirm-inline';
  box.style.cssText = 'background:#FFF3F3;border:1px solid #A63A2E;border-radius:6px;padding:10px 12px;margin-top:8px;display:flex;flex-wrap:wrap;gap:8px;align-items:center;';
  const span = document.createElement('span');
  span.style.cssText = 'font-size:.84rem;flex:1 1 100%;';
  span.textContent = texto;
  box.appendChild(span);
  const fechar = ()=>{ box.remove(); anchor.style.display = ''; };
  acoes.forEach(a=>{
    const b = document.createElement('button');
    b.className = 'btn btn-danger btn-small'; b.textContent = a.label;
    b.onclick = ()=>{ fechar(); a.fn(); };
    box.appendChild(b);
  });
  const c = document.createElement('button');
  c.className = 'btn btn-outline btn-small'; c.textContent = 'Cancelar'; c.onclick = fechar;
  box.appendChild(c);
  anchor.style.display = 'none';
  anchor.parentNode.insertBefore(box, anchor.nextSibling);
}

async function loadRoster(){
  try{
    const r = await window.storage.get('ibnv-roster', true);
    let current = r ? JSON.parse(r.value) : [];
    const existingIds = new Set(current.map(m=>m.id));
    const toAdd = SEED_ROSTER.filter(s=>!existingIds.has(s.id));
    if(toAdd.length){
      current = current.concat(toAdd);
    }
    current = current.map(m=>({senha:'', ...m}));
    if(toAdd.length || !r){
      await window.storage.set('ibnv-roster', JSON.stringify(current), true);
    }
    return current;
  }
  catch(e){ return SEED_ROSTER.map(m=>({senha:'', ...m})); }
}

      if(!ex){ current.push(s); changed = true; }
      else if(ex.seedV!==s.seedV){
        ex.nome = s.nome; ex.telefone = s.telefone; ex.criadoEm = s.criadoEm;
        ex.coordenacao = ex.coordenacao || {};
        ex.coordenacao.observacao = s.coordenacao.observacao;
        ex.seedV = s.seedV; changed = true;
      }
    });
    if(changed) await window.storage.set('ibnv-people', JSON.stringify(current), true);
    return current;
  }
  catch(e){ return SEED_PEOPLE.slice(); }
}

function daysSince(d){ if(!d) return null; const ms = Date.now() - new Date(d).getTime(); return Math.floor(ms/86400000); }
const SEV = {verde:0, amarelo:1, laranja:2, vermelho:3};

function ausenciaNivel(dias){
  if(dias===null) return 'verde';
  if(dias>=60) return 'vermelho';
  if(dias>=30) return 'laranja';
  if(dias>=15) return 'amarelo';
  return 'verde';
}
function necessidadeNivel(n){
  return {Nenhuma:'verde', Leve:'amarelo', Relevante:'laranja', Grave:'vermelho'}[n] || 'verde';
}
function pessoaAlerta(p){
  const datas = [p.cuidado.ultimaParticipacao, p.cuidado.ultimoContato, p.consolidacao.ultimoContato, p.discipulado.ultimoEncontro, p.criadoEm]
    .filter(Boolean).map(d=>new Date(d).getTime());
  const maisRecente = datas.length ? new Date(Math.max(...datas)) : new Date(p.criadoEm);
  const dias = daysSince(maisRecente);
  const nAus = ausenciaNivel(dias);
  const nNec = necessidadeNivel(p.cuidado.necessidade);
  return SEV[nAus] >= SEV[nNec] ? nAus : nNec;
}
function pessoaPendencias(p){
  const pend = [];
  const diasCriacao = daysSince(p.criadoEm);
  if(p.status==='Visitante' && !p.consolidacao.ultimoContato && diasCriacao!==null && diasCriacao>2)
    pend.push('Visitante sem contato após 48h');
  if(pessoaAlerta(p)==='vermelho' && !p.cuidado.encaminhamentoPastoral)
    pend.push('Alerta vermelho sem encaminhamento pastoral');
  ['consolidacao','discipulado','cuidado'].forEach(sec=>{
    if(p[sec].prazo && new Date(p[sec].prazo) < new Date())
      pend.push('Prazo vencido — '+teamLabel(sec));
  });
  if(p.status==='Em discipulado'){
    const d = daysSince(p.discipulado.ultimoEncontro);
    if(d!==null && d>30) pend.push('Discipulado sem atualização há 30+ dias');
  }
  return pend;
}

  return String(s||'').normalize('NFD').replace(/[\u0300-\u036f]/g,'').toLowerCase()

}

  if(canManageRoster() || isLeaderNivel()) tabs.push({key:'organograma', label:'Organograma'});
  if(canManageRoster()) tabs.push({key:'config', label:'Configurações'});
  if(isKidsOnly()) tabs.splice(0, tabs.length, {key:'kids', label:'Kids'});
  if(!tabs.some(t=>t.key===activeTab)) activeTab = tabs[0].key;
  tabs.forEach(t=>{
    const b = document.createElement('button');
    b.textContent = t.label;
    if(activeTab===t.key) b.className='active';
    b.onclick = ()=>{ activeTab=t.key; expandedId=null; renderMain(); };
    tabsWrap.appendChild(b);
  });
  app.appendChild(tabsWrap);

  const content = document.createElement('div');
  content.id = 'tabContent';
  app.appendChild(content);

  if(activeTab==='painel') renderPainel(content);
  else if(activeTab==='pessoas') renderPessoas(content);
  else if(activeTab==='formulario') renderFormulario(content);
  else if(activeTab==='relatorio' && canSeeReport()) renderRelatorio(content);
  else if(activeTab==='minhaequipe' && isLeaderNivel()) renderMinhaEquipe(content);
  else if(activeTab==='kids' && canSeeKids()) renderKids(content);

  else if(activeTab==='equipe' && canManageRoster()) renderEquipe(content);
  else if(activeTab==='organograma' && (canManageRoster() || isLeaderNivel())) renderOrganograma(content);
  else if(activeTab==='config' && canManageRoster()) renderConfig(content);
}

function idadeDe(nasc){
  if(!nasc) return '';
  const n = new Date(nasc+'T12:00:00'); if(isNaN(n)) return '';
  const h = new Date();
  let anos = h.getFullYear()-n.getFullYear();
  let meses = h.getMonth()-n.getMonth();
  if(h.getDate()<n.getDate()) meses--;
  if(meses<0){ anos--; meses += 12; }
  if(anos<0) return '';
  return anos<2 ? (anos*12+meses)+' meses' : anos+' anos';
}

function kidGet(k,path){ return path.split('.').reduce((o,key)=>o==null?o:o[key],k); }
function kidSet(k,path,val){
  const a = path.split('.'); let o = k;
  for(let i=0;i<a.length-1;i++) o = o[a[i]];
  o[a[a.length-1]] = val;
}
function kidTemAtencao(k){
  const s=k.saude, e=k.especiais, r=k.regulacao;
  return !!(s.alergias||s.restricao||s.medicacao==='Sim'||(s.banheiro&&s.banheiro!=='Não')||s.outro||
    e.opcoes.length||e.outra||r.incomoda.length||r.incomodaOutro||r.acalma||r.sinais||r.fazerEvitar);
}
function kidChips(k){
  const c=[], s=k.saude, e=k.especiais, r=k.regulacao;
  if(s.alergias) c.push('Alergia');
  if(s.restricao) c.push('Restrição alimentar');
  if(s.medicacao==='Sim') c.push('Medicação no culto');
  if(s.banheiro && s.banheiro!=='Não') c.push(s.banheiro==='Usa fralda'?'Fralda':'Ajuda no banheiro');
  if(e.opcoes.length||e.outra) c.push('Necessidade especial');
  if(r.incomoda.length||r.incomodaOutro||r.acalma||r.sinais||r.fazerEvitar) c.push('Regulação');
  return c;
}
function kidPendencias(k){
  const p=[];
  if(!k.bastao.enviado) p.push('Passagem de bastão à Recepção/Consolidação não registrada');
  const dias = daysSince(k.visita.data ? k.visita.data+'T12:00:00' : k.criadoEm);
  if(k.bastao.enviado && k.autorizacoes.contato && !k.acompanhamento.contatoData && dias!==null && dias>2)
    p.push('Sem contato com a família após 48h');
  if(kidTemAtencao(k) && !k.equipe.liderancaInformada) p.push('Saúde/regulação ainda não passada à liderança do Kids');
  if(kidTemAtencao(k) && !k.autorizacoes.usoInformacoes) p.push('Autorização de uso das informações não marcada');
  return p;
}

function kfield(k,label,path,type,full){
  const v = kidGet(k,path)||'';
  if(type==='textarea') return `<div class="full"><label>${label}</label><textarea data-k="${path}">${esc(v)}</textarea></div>`;
  return `<div class="${full?'full':''}"><label>${label}</label><input type="${type||'text'}" data-k="${path}" value="${esc(v)}"></div>`;
}
function ksel(k,label,path,options){
  const v = kidGet(k,path)||'';
  return `<div><label>${label}</label><select data-k="${path}">${options.map(o=>`<option value="${esc(o)}" ${v===o?'selected':''}>${o||'—'}</option>`).join('')}</select></div>`;
}
function kchk(k,path,options){
  const arr = kidGet(k,path)||[];
  return options.map(o=>`<label class="checkline" style="margin-top:6px;"><input type="checkbox" data-arr="${path}" value="${esc(o)}" ${arr.includes(o)?'checked':''}> ${esc(o)}</label>`).join('');
}
function kbool(k,label,path){
  return `<label class="checkline"><input type="checkbox" data-k="${path}" data-bool="1" ${kidGet(k,path)?'checked':''}> ${label}</label>`;
}

async function kidEnviarBastao(k){
  const nome = (k.responsavel.nome||'').trim();
  if(!nome){ alert('Preencha o nome do responsável antes de passar o bastão.'); return false; }
  const cri = (k.crianca.nome||'').trim()||'criança';
  const idade = idadeDe(k.crianca.nascimento);
  const obs = 'Família com criança acolhida no Kids: '+cri+(idade?' ('+idade+')':'')+'. Citar no contato que a criança foi acolhida no Kids.';
  let p = people.find(x=>normName(x.nome)===normName(nome));
  if(p){
    p.coordenacao = p.coordenacao || {observacao:'', encaminhamentoPastoral:false};
    p.coordenacao.observacao = (p.coordenacao.observacao ? p.coordenacao.observacao+' | ' : '')+obs;
  } else {
    p = novaPessoa('kids');
    p.nome = nome; p.telefone = k.responsavel.telefone||''; p.autorizaContato = !!k.autorizacoes.contato;
    const d = k.visita.data ? new Date(k.visita.data+'T12:00:00') : new Date();
    p.criadoEm = isNaN(d) ? new Date().toISOString() : d.toISOString();
    p.coordenacao.observacao = obs;
    p.consolidacao.proximoPasso = 'Contato citando a criança acolhida no Kids';
    people.unshift(p);
  }
  await savePeople(people);
  k.bastao = {enviado:true, pessoaId:p.id, em:new Date().toISOString()};
  await saveKids(kids);
  return true;
}

const DIAC_AREAS = ['Portaria','Recepção ao culto','Ofertório','Santa Ceia','Estacionamento e segurança','Visitação','Apoio ao púlpito'];

function canSeeDiaconato(){
  return !!currentUser && (currentUser.role==='coordenacao' || (currentUser.role==='membro' && currentUser.team==='diaconato'));
}
function canManageDiaconato(){
  return !!currentUser && (currentUser.role==='coordenacao' ||
    (currentUser.role==='membro' && currentUser.team==='diaconato' && (currentUser.nivel==='Líder' || currentUser.nivel==='Adjunto')));
}
async function loadDiacList(chave){
  try{ const r = await safeGet(chave); return r && r.value ? JSON.parse(r.value) : []; }
  catch(e){ return []; }
}
async function saveDiacList(chave, lista){
  try{ await window.storage.set(chave, JSON.stringify(lista), true); }
  catch(e){ console.error('Erro ao salvar '+chave, e); }
}
const saveDiacos = ()=>saveDiacList('ibnv-diac-diaconos', diacos);
const saveDiacEscalas = ()=>saveDiacList('ibnv-diac-escalas', diacEscalas);
const saveDiacCeias = ()=>saveDiacList('ibnv-diac-ceias', diacCeias);
async function ensureDiac(force){
  if(!diacLoaded || force){
    diacos = await loadDiacList('ibnv-diac-diaconos');
    diacEscalas = await loadDiacList('ibnv-diac-escalas');
    diacCeias = await loadDiacList('ibnv-diac-ceias');
    diacLoaded = true;
  }
}
function diacId(p){ return p+Date.now()+Math.floor(Math.random()*1000); }
function novoDiacono(situacao){
  return {
    id:diacId('d'), criadoEm:new Date().toISOString(), nome:'', telefone:'', nascimento:'', situacao, consagracao:'',
    areas:[], observacao:'',
    treinamento:{inicio: situacao==='Em treinamento' ? hojeKey() : '', previsao:'', responsavel:'',
      etapas: DIAC_ETAPAS_PADRAO.map(n=>({nome:n, feito:false})), observacao:''}
  };
}
function novaEscala(){
  return {id:diacId('e'), criadoEm:new Date().toISOString(), data:hojeKey(), culto:DIAC_CULTOS[0], observacao:'', itens:[]};
}
function novaCeia(){
  return {id:diacId('c'), criadoEm:new Date().toISOString(), data:hojeKey(), culto:DIAC_CULTOS[1], ministrante:'', estimativa:'',
    status:'Agendada', observacao:'', preparoFeito:[],
    itens: DIAC_FUNCOES_CEIA.map(f=>({funcao:f, diaconoId:'', diaconoNome:'', confirmado:false}))};
}
function diacAtivos(){
  return diacos.filter(d=>d.situacao==='Ativo').sort((a,b)=>String(a.nome).localeCompare(String(b.nome),'pt-BR'));
}
function diacProgresso(d){
  const e = (d.treinamento && d.treinamento.etapas) || [];
  return e.length ? Math.round(e.filter(x=>x.feito).length/e.length*100) : 0;
}
function diaSemana(key){
  const d = new Date(key+'T12:00:00');
  return isNaN(d) ? '' : d.toLocaleDateString('pt-BR',{weekday:'long'});
}
function diacConflitos(rec){
  const vistos = {};
  (rec.itens||[]).forEach(it=>{ if(it.diaconoId) vistos[it.diaconoId] = (vistos[it.diaconoId]||0)+1; });
  diacEscalas.concat(diacCeias).forEach(o=>{
    if(o.id!==rec.id && o.data===rec.data && o.culto===rec.culto)
      (o.itens||[]).forEach(it=>{ if(it.diaconoId && vistos[it.diaconoId]!==undefined) vistos[it.diaconoId]++; });
  });
  return Object.keys(vistos).filter(id=>vistos[id]>1).map(id=>{ const d = diacos.find(x=>x.id===id); return d ? d.nome : 'Diácono'; });
}
function diacTexto(e, tipo){
  let t = '*'+(tipo==='ceia'?'Santa Ceia':'Escala de serviço')+' — '+fmtDayBR(e.data)+' ('+diaSemana(e.data)+')*\n'+e.culto;
  if(tipo==='ceia' && e.ministrante) t += '\nMinistrante: '+e.ministrante;
  (e.itens||[]).forEach(it=>{ t += '\n• '+it.funcao+': '+(it.diaconoNome||'(a definir)')+(it.confirmado?' ✅':''); });
  if(e.observacao) t += '\nObs.: '+e.observacao;
  return t;
}
async function renderDiaconato(content){
  if(!canSeeDiaconato()){ content.innerHTML = ''; return; }
  content.innerHTML = '<div class="empty-note">Carregando...</div>';
  await ensureDiac();
  drawDiac(content);
}
function drawDiac(content){
  const subs = [['diaconos','Diáconos'],['treino','Em treinamento'],['escalas','Escalas de serviço'],['ceia','Santa Ceia']];
  const perfil = canManageDiaconato()
    ? 'Liderança do Diaconato: você cadastra, edita e exclui diáconos, escalas e Santa Ceia.'
    : 'Diaconato: você consulta e edita escalas, Santa Ceia e treinamento. Cadastrar e excluir é com a liderança.';
  content.innerHTML = `
    <div class="banner">${perfil}</div>
    <nav class="tabs">${subs.map(s=>`<button data-dsub="${s[0]}" class="${diacView===s[0]?'active':''}">${s[1]}</button>`).join('')}</nav>
    <div id="diacBody"></div>`;
  content.querySelectorAll('[data-dsub]').forEach(b=>{
    b.onclick = ()=>{ diacView = b.getAttribute('data-dsub'); diacExpanded = null; drawDiac(content); };
  });
  const body = content.querySelector('#diacBody');
  if(diacView==='diaconos' || diacView==='treino') renderDiacPessoas(body, content, diacView==='treino');
  else renderDiacEventos(body, content, diacView);
}

  if(add) add.onclick = async ()=>{
    const d = novoDiacono(treino ? 'Em treinamento' : 'Ativo');

  };

  body.querySelector('#dRefresh').onclick = async ()=>{ await ensureDiac(true); drawDiac(content); };
  renderDiacPessoasList(lista, content, treino);
}
function renderDiacPessoasList(listEl, content, treino){
  const q = normName(diacSearch);
  const itens = diacos.filter(d=>{
    if(treino ? d.situacao!=='Em treinamento' : d.situacao==='Em treinamento') return false;
    if(!treino && diacFiltro!=='todos' && d.situacao!==diacFiltro) return false;
    return !q || normName(d.nome).includes(q);
  }).sort((a,b)=>String(a.nome).localeCompare(String(b.nome),'pt-BR'));
  if(!itens.length){ listEl.innerHTML = `<div class="empty-note">${treino?'Ninguém em treinamento no momento.':'Nenhum diácono encontrado.'}</div>`; return; }
  listEl.innerHTML = '';
  itens.forEach(d=>{
    const row = document.createElement('div'); row.className = 'person-row';
    const prog = diacProgresso(d);
    row.innerHTML = `
      <div class="person-head">
        <div>
          <div class="name">${esc(d.nome||'(sem nome)')}</div>
          <div class="meta">${esc(d.situacao)}${d.telefone?' · '+esc(d.telefone):''}${treino && d.treinamento.responsavel?' · Tutor: '+esc(d.treinamento.responsavel):''}</div>
          ${treino ? `<div class="phase-track" style="height:10px;margin-top:6px;max-width:260px;"><div class="phase-fill" style="width:${prog}%;background:var(--wine);"></div></div><div class="hint" style="margin:2px 0 0;">${prog}% das etapas concluídas</div>` : ''}
          <div>${(d.areas||[]).map(a=>'<span class="chip">'+esc(a)+'</span>').join('')}</div>
        </div>
        <span class="pill">${diacExpanded===d.id?'fechar ▲':'abrir ▼'}</span>
      </div>
      <div class="person-body ${diacExpanded===d.id?'open':''}"></div>`;
    listEl.appendChild(row);
    row.querySelector('.person-head').onclick = ()=>{ diacExpanded = diacExpanded===d.id ? null : d.id; renderDiacPessoasList(listEl, content, treino); };
    if(diacExpanded===d.id) renderDiaconoEditor(row.querySelector('.person-body'), d, listEl, content, treino);
  });
}

function canSeeReport(){
  const t = myTeamKey();
  if(t==='diaconato') return false;
  return t==='coordenacao' || t==='inchurch' || t==='recepcao' || isLeaderNivel();

let reportDate = null;
function localDayKey(d){
  const x = new Date(d);
  return x.getFullYear()+'-'+String(x.getMonth()+1).padStart(2,'0')+'-'+String(x.getDate()).padStart(2,'0');
}
function fmtDayBR(key){ const a = key.split('-'); return a[2]+'/'+a[1]+'/'+a[0]; }
function shiftDay(key, delta){
  const a = key.split('-').map(Number);
  return localDayKey(new Date(a[0], a[1]-1, a[2]+delta));
}
function esc(s){
  return String(s==null?'':s).replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/"/g,'&quot;');
}
function reportList(dayKey){
  return people.filter(p=>localDayKey(p.criadoEm)===dayKey)
    .sort((a,b)=>new Date(a.criadoEm)-new Date(b.criadoEm));
}
function alertaLabel(p){ const a = pessoaAlerta(p); return a.charAt(0).toUpperCase()+a.slice(1); }
function buildReportCsv(list){
  const q = v => '"'+String(v==null?'':v).replace(/"/g,'""')+'"';
  const head = ['Nº','Nome','Telefone / WhatsApp','Data do cadastro','Hora','Status','Cadastrado por','Autoriza contato','Alerta','Pendências','Observação'];
  const lines = [head.map(q).join(';')];
  list.forEach((p,i)=>{
    const dt = new Date(p.criadoEm);
    lines.push([
      i+1, p.nome, p.telefone||'', fmtDayBR(localDayKey(dt)),
      p.seedV ? '—' : String(dt.getHours()).padStart(2,'0')+':'+String(dt.getMinutes()).padStart(2,'0'),
      p.status, teamLabel(p.criadoPor), p.autorizaContato?'Sim':'Não',
      alertaLabel(p), pessoaPendencias(p).join(' | '),
      (p.coordenacao && p.coordenacao.observacao) || ''
    ].map(q).join(';'));
  });
  return '\ufeff'+lines.join('\r\n');
}
function buildReportText(list, dayKey){
  let t = '*IBNV Conectar — Cadastros de '+fmtDayBR(dayKey)+'*\nTotal: '+list.length+'\n';
  list.forEach((p,i)=>{ t += '\n'+(i+1)+'. '+p.nome+' — '+(p.telefone||'sem telefone')+' ('+p.status+')'; });
  return t;
}
function lerArquivoTexto(file){
  return new Promise((resolve, reject)=>{
    const fr = new FileReader();
    fr.onload = ()=>resolve(String(fr.result));
    fr.onerror = ()=>reject(fr.error);
    fr.readAsText(file);
  });
}
function downloadText(filename, text, mime){
  const blob = new Blob([text], {type:mime});
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url; a.download = filename;
  document.body.appendChild(a); a.click(); a.remove();
  URL.revokeObjectURL(url);
}
function copyText(text){
  const fallback = ()=>{
    const ta = document.createElement('textarea');
    ta.value = text; document.body.appendChild(ta); ta.select();
    let ok = false; try{ ok = document.execCommand('copy'); }catch(e){}
    ta.remove();
    alert(ok ? 'Resumo copiado.' : 'Não foi possível copiar automaticamente.');
  };
  if(navigator.clipboard && navigator.clipboard.writeText){
    navigator.clipboard.writeText(text).then(()=>alert('Resumo copiado.')).catch(fallback);
  } else fallback();
}

function downloadBlankForm(teamKey){

function applyFormData(person, teamKey, data){
  const fields = FORM_FIELDS[teamKey] || [];
  fields.forEach(f=>{
    if(f.target==='nome') return;
    if(!(f.target in data)) return;
    const val = f.type==='checkbox' ? !!data[f.target] : data[f.target];
    const path = f.target.split('.');
    let obj = person;
    for(let i=0;i<path.length-1;i++) obj = obj[path[i]];
    obj[path[path.length-1]] = val;
  });
}

async function handleFormUpload(file, teamKey){
  let data;
  try{
    const text = await file.text();
    data = JSON.parse(text);
  } catch(e){
    alert('Não foi possível ler este arquivo. Verifique se é um .json gerado pelo formulário do Conectar.');
    return;
  }
  const nome = (data.nome || '').trim();
  if(!nome){ alert('Formulário inválido: falta o nome da pessoa.'); return; }
  let person = people.find(p=>p.nome.toLowerCase()===nome.toLowerCase());
  if(!person){
    const podeCriar = teamKey==='recepcao' || teamKey==='inchurch' || teamKey==='coordenacao';
    if(!podeCriar){
      alert('"'+nome+'" ainda não está cadastrado(a). Peça para a Recepção ou o InChurch cadastrarem essa pessoa primeiro, ou confira se o nome está exatamente igual.');
      return;
    }
    person = novaPessoa(teamKey);
    person.nome = nome;
    people.unshift(person);
  }
  applyFormData(person, teamKey, data);
  try{ await savePeople(people); }
  catch(e){ console.error('Erro ao salvar pessoas', e); }
  alert('Formulário de "'+nome+'" processado e lançado no cadastro com sucesso.');
  renderMain();
}

function renderPainel(content){
  const base = visiblePeople();
  const counts = {verde:0,amarelo:0,laranja:0,vermelho:0};
  let pendCount=0, encamCount=0;
  base.forEach(p=>{
    counts[pessoaAlerta(p)]++;
    if(pessoaPendencias(p).length) pendCount++;
    if(p.cuidado.encaminhamentoPastoral || p.coordenacao.encaminhamentoPastoral) encamCount++;
  });

  const PHASES = [
    {status:'Visitante', label:'Visitante', team:'Recepção'},
    {status:'Em consolidação', label:'Em consolidação', team:'Consolidação'},
    {status:'Novo convertido', label:'Novo convertido', team:'Consolidação'},
    {status:'Em discipulado', label:'Em discipulado', team:'Discipulado'},
    {status:'Preparação para batismo', label:'Preparação para batismo', team:'Discipulado'},
    {status:'Integrado', label:'Integrado', team:'InChurch'},
    {status:'Membro', label:'Membro', team:'Fase final'},
  ];
  const OUTROS = ['Sem retorno','Visitante eventual','Pertence a outra igreja','Em cuidado','Afastado'];
  const phaseCounts = PHASES.map(f=>base.filter(p=>p.status===f.status).length);
  const outrosCount = base.filter(p=>OUTROS.includes(p.status)).length;
  const maxCount = Math.max(1, ...phaseCounts);
  const n = PHASES.length;
  const barsHtml = PHASES.map((f,i)=>{
    const c = phaseCounts[i];
    const pct = Math.round((c/maxCount)*100);
    const hue = 8 + Math.round((i/(n-1))*4); // leve variação de tom, mantendo a família do vermelho
    const light = 74 - Math.round((i/(n-1))*34); // mais claro no início, mais escuro na fase final
    return `
      <div class="phase-row">
        <div class="phase-label">${f.label}<span class="phase-team">${f.team}</span></div>
        <div class="phase-track">
          <div class="phase-fill" style="width:${Math.max(pct,c>0?4:0)}%;background:hsl(${hue} 68% ${light}%);"></div>
        </div>
        <div class="phase-count">${c}</div>
      </div>
    `;
  }).join('');

  content.innerHTML = `
    <div class="stat-row">
      <div class="stat"><div class="num">${base.length}</div><div class="lbl">Pessoas acompanhadas</div></div>
      <div class="stat"><div class="num"><span class="dot verde"></span>${counts.verde}</div><div class="lbl">Verde</div></div>
      <div class="stat"><div class="num"><span class="dot amarelo"></span>${counts.amarelo}</div><div class="lbl">Amarelo</div></div>
      <div class="stat"><div class="num"><span class="dot laranja"></span>${counts.laranja}</div><div class="lbl">Laranja</div></div>
      <div class="stat"><div class="num"><span class="dot vermelho"></span>${counts.vermelho}</div><div class="lbl">Vermelho</div></div>
    </div>
    <div class="stat-row">
      <div class="stat"><div class="num">${pendCount}</div><div class="lbl">Com pendências</div></div>
      <div class="stat"><div class="num">${encamCount}</div><div class="lbl">Encaminhados à liderança pastoral</div></div>
    </div>

    <div class="card" style="margin-top:22px;">
      <h3 style="margin-top:0;">Evolução do projeto por fase</h3>
      <p class="hint" style="margin-bottom:16px;">Quantas pessoas estão em cada etapa do fluxo, da Recepção até a fase final (Membro).</p>
      <div class="phase-chart">${barsHtml}</div>
      ${outrosCount>0 ? `<p class="hint" style="margin-top:14px;">+ ${outrosCount} pessoa${outrosCount===1?'':'s'} em situações à parte do fluxo principal (sem retorno, afastado, outra igreja etc.)</p>` : ''}
    </div>

    <p class="hint" style="margin-top:18px;">Sua área de atuação: <b>${myTeamKey()==='coordenacao' ? 'Coordenação (acesso completo)' : teamLabel(myTeamKey())}</b>. Vá para a aba Pessoas para registrar ou atualizar alguém.</p>
  `;
}

function renderPessoasList(){
  const list = document.getElementById('peopleList');
  if(!list) return;
  let filtered = visiblePeople().filter(p=>{
    if(searchTerm && !p.nome.toLowerCase().includes(searchTerm.toLowerCase())) return false;
    if(filterAlerta!=='todos' && pessoaAlerta(p)!==filterAlerta) return false;
    if(onlyPendencias && pessoaPendencias(p).length===0) return false;
    return true;
  });
  if(filtered.length===0){
    list.innerHTML = '<div class="empty-note">Nenhuma pessoa encontrada com esses filtros.</div>';
    return;
  }
  list.innerHTML = '';
  filtered.forEach(p=>{
    const row = document.createElement('div');
    row.className='person-row';
    const pend = pessoaPendencias(p);
    row.innerHTML = `
      <div class="person-head" data-id="${p.id}">
        <div>
          <div class="name"><span class="dot ${pessoaAlerta(p)}"></span>${p.nome||'(sem nome)'}</div>
          <div class="meta">${p.status} · cadastrado em ${fmtDate(p.criadoEm)} por ${teamLabel(p.criadoPor)}</div>
          ${pend.length? `<div class="flag">${pend.join(' · ')}</div>` : ''}
        </div>
        <span class="pill">${expandedId===p.id ? 'fechar ▲' : 'abrir ▼'}</span>
      </div>
      <div class="person-body ${expandedId===p.id?'open':''}" id="body-${p.id}"></div>
    `;
    list.appendChild(row);
    row.querySelector('.person-head').onclick = ()=>{
      expandedId = expandedId===p.id ? null : p.id;
      renderPessoasList();
    };
    if(expandedId===p.id){
      renderPersonDetail(document.getElementById('body-'+p.id), p);
    }
  });
}

            ${TEAMS.map(t=>`<option value="${t.key}" ${r.team===t.key?'selected':''}>${t.label}</option>`).join('')}
          </select>
        </td>
        <td>
          <select data-id="${r.id}" class="roleSelect">
            ${['Líder','Adjunto','Integrante'].map(fn=>`<option value="${fn}" ${(r.role||'Integrante')===fn?'selected':''}>${fn}</option>`).join('')}
          </select>
        </td>
        <td>
          <div style="display:flex;flex-direction:column;gap:3px;">
            <span class="pill">${r.senha ? '✓ própria' : 'usa a padrão'}</span>
            <input type="password" data-id="${r.id}" class="senhaInput" placeholder="Nova senha">
          </div>
        </td>
        <td><input type="checkbox" class="discCheck" data-id="${r.id}" ${r.team==='discipulado' ? 'checked disabled title="Equipe de Discipulado já pode entrar como discipulador"' : (r.discipulador?'checked':'')}></td>
        <td><button class="btn btn-danger btn-small" data-remove="${r.id}">Remover</button></td>
      `;
      body.appendChild(tr);
    });
  }
  content.querySelectorAll('.teamSelect').forEach(sel=>{
    sel.onchange = async ()=>{
      const item = roster.find(r=>r.id===sel.getAttribute('data-id'));
      item.team = sel.value;
      await saveRoster(roster);
    };
  });
  content.querySelectorAll('.roleSelect').forEach(sel=>{
    sel.onchange = async ()=>{
      const item = roster.find(r=>r.id===sel.getAttribute('data-id'));
      item.role = sel.value;
      await saveRoster(roster);
    };
  });
  content.querySelectorAll('.discCheck').forEach(cb=>{
    cb.onchange = async ()=>{
      const item = roster.find(r=>r.id===cb.getAttribute('data-id'));
      item.discipulador = cb.checked;
      await saveRoster(roster);
    };
  });
  content.querySelectorAll('.senhaInput').forEach(inp=>{
    inp.onchange = async ()=>{
      const val = inp.value.trim();
      if(!val) return;
      if(val.length<4){ alert('Use ao menos 4 caracteres.'); inp.value=''; return; }
      const item = roster.find(r=>r.id===inp.getAttribute('data-id'));
      item.senha = val;
      await saveRoster(roster);
      renderEquipe(content);
    };
  });
  content.querySelectorAll('[data-remove]').forEach(btn=>{
    btn.onclick = ()=>{
      const id = btn.getAttribute('data-remove');
      pedirConfirmacao(btn, 'Remover este integrante da equipe?', async ()=>{
        roster = roster.filter(r=>r.id!==id);
        await saveRoster(roster);
        renderEquipe(content);
      });
    };
  });
  content.querySelector('#addForm').onsubmit = async (e)=>{
    e.preventDefault();
    const name = content.querySelector('#newName').value.trim();
    const team = content.querySelector('#newTeam').value;
    const role = content.querySelector('#newRole').value;
    const senha = content.querySelector('#newSenha').value.trim();
    const disc = content.querySelector('#newDisc').checked;
    if(!name || (senha.length>0 && senha.length<4)) return;
    if(roster.some(r=>r.name.toLowerCase()===name.toLowerCase())){
      alert('Já existe um integrante com esse nome. Ajuste o nome ou edite a equipe dele na lista abaixo.');
      return;
    }
    roster.push({id:'m'+Date.now()+Math.floor(Math.random()*1000), name, team, role, senha, discipulador: disc});
    await saveRoster(roster);
    renderEquipe(content);
  };
}

    downloadText('backup-ibnv-conectar-'+localDayKey(new Date())+'.json', JSON.stringify(dados,null,2), 'application/json');

    const file = ev.target.files && ev.target.files[0];
    if(!file) return;
    let dados;
    try{ dados = JSON.parse(await lerArquivoTexto(file)); }
    catch(e){ toast('Arquivo inválido: não consegui ler esse .json.'); return; }
    if(!dados || typeof dados!=='object'){ toast('Arquivo inválido.'); return; }
    await ensureKids(); await ensureDiac();

      checklists: somar(kidsChecks, dados.kidsChecks),

    if(n.checklists) await saveKidsChecks(kidsChecks);
    if(n.diaconos) await saveDiacos();
    if(n.escalas) await saveDiacEscalas();
    if(n.ceias) await saveDiacCeias();
    ev.target.value = '';
    toast('Importado: '+n.pessoas+' pessoa(s), '+n.integrantes+' integrante(s), '+n.criancas+' criança(s), '+n.checklists+' checklist(s), '+n.diaconos+' diácono(s), '+n.escalas+' escala(s), '+n.ceias+' Santa(s) Ceia(s).');
